🔨 구체적 수정 방법

수정 1️⃣: 상수 범위 정의 (generator.py)

문제: 매개변수 범위가 주석에만 있고 코드에서 강제되지 않음

❌ BEFORE
@dataclass class WeightConfig: min_vph_threshold: float = 50.0 # 주석만 있음: 10-200 min_std_floor: float = 5.0 # 주석만 있음: 1-20 red_ocean_weight: float = 0.5 # 주석만 있음: 0.1-2.0 red_ocean_cap: float = 1.5 # 주석만 있음: 1.1-3.0
✅ AFTER
from dataclasses import dataclass, field @dataclass class WeightConfig: # Phase 1 min_vph_threshold: float = field(default=50.0) min_std_floor: float = field(default=5.0) # Phase 2 red_ocean_weight: float = field(default=0.5) red_ocean_cap: float = field(default=1.5) def __post_init__(self): """Validate ranges after initialization.""" RANGES = { 'min_vph_threshold': (10.0, 200.0), 'min_std_floor': (1.0, 20.0), 'red_ocean_weight': (0.1, 2.0), 'red_ocean_cap': (1.1, 3.0), } for param, (min_val, max_val) in RANGES.items(): val = getattr(self, param) if not (min_val <= val <= max_val): raise ValueError( f"{param}={val} out of range [{min_val}, {max_val}]" )
✅ 안정성 향상 | autoresearch 방식 따라가기

수정 2️⃣: 시드 고정으로 재현성 보장 (orchestrator.py)

문제: LLM 응답의 비결정성 → 같은 상황에서 다른 가중치 제안

❌ BEFORE
def main() -> None: parser = argparse.ArgumentParser() parser.add_argument("--cycles", type=int, default=1) args = parser.parse_args() # 시드 설정 없음 → 재현 불가능 current = load_best_weights()
✅ AFTER
import random import numpy as np def main() -> None: parser = argparse.ArgumentParser() parser.add_argument("--cycles", type=int, default=1) parser.add_argument("--seed", type=int, default=42) args = parser.parse_args() # autoresearch 방식: 시드 고정 SEED = args.seed random.seed(SEED) np.random.seed(SEED) log.info(f"Random seed: {SEED} (for reproducibility)") current = load_best_weights() current.random_seed = SEED
✅ 완벽한 재현성 | autoresearch의 Seed 42 방식 채택
